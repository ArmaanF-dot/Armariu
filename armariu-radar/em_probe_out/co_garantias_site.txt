    document.addEventListener('DOMContentLoaded', function () {
        document.querySelectorAll('video.video-no-autoplay').forEach(function (v) {
            // 1) Quitar atributo autoplay si existe (por si acaso)
            v.removeAttribute('autoplay');
            // 2) Asegurar preload y controls
            v.preload = 'metadata';
            v.controls = true;

            // 3) Pausar inmediatamente para neutralizar reproducciones automáticas
            try { v.pause(); } catch (e) { }

            // 4) Si hay intentos programáticos de play, bloquearlos salvo que sea interacción del usuario

            let userInteracted = false;
            v.addEventListener('click', function () { userInteracted = true; });
            v.addEventListener('touchstart', function () { userInteracted = true; });

                // permitir play sólo si el usuario interactuó

                    return originalPlay.apply(v, arguments);
                }
                // si es intento programático, ignorar silenciosamente y devolver Promise resuelta
                return Promise.resolve();
            };

            v.addEventListener('canplay', function () { try { v.pause(); } catch (e) { } });
        });
    });
